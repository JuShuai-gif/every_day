cmake_minimum_required(VERSION 3.20)
project(lidar_filter LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(EDGE_SANITIZE "Address and undefined-behavior sanitizers" OFF)
option(EDGE_TSAN "Thread sanitizer in a separate build" OFF)
if(EDGE_SANITIZE AND EDGE_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
include(cmake/edge-options.cmake)
find_package(Threads REQUIRED)
file(GLOB PROJECT_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(lidar_filter ${PROJECT_SOURCES})
target_include_directories(lidar_filter PRIVATE include)
target_link_libraries(lidar_filter PRIVATE Threads::Threads)
edge_configure_target(lidar_filter)
target_compile_options(lidar_filter PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-omit-frame-pointer)
if(EDGE_SANITIZE)
  target_compile_options(lidar_filter PRIVATE -O1 -g -fsanitize=address,undefined)
  target_link_options(lidar_filter PRIVATE -fsanitize=address,undefined)
endif()
if(EDGE_TSAN)
  target_compile_options(lidar_filter PRIVATE -O1 -g -fsanitize=thread)
  target_link_options(lidar_filter PRIVATE -fsanitize=thread)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME correctness COMMAND lidar_filter --test)
  add_test(NAME benchmark_smoke COMMAND lidar_filter --mode all --samples 2 --warmup 1)
  add_test(NAME reject_unknown_option COMMAND lidar_filter --invalid)
  add_test(NAME reject_zero_samples COMMAND lidar_filter --samples 0)
  set_tests_properties(reject_unknown_option reject_zero_samples PROPERTIES WILL_FAIL TRUE)
  set_tests_properties(correctness benchmark_smoke PROPERTIES TIMEOUT 90)
endif()
