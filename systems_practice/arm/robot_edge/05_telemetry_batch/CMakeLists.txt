cmake_minimum_required(VERSION 3.20)
project(telemetry_batch LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(EDGE_SANITIZE "Address and undefined-behavior sanitizers" OFF)
option(EDGE_TSAN "Thread sanitizer in a separate build" OFF)
if(EDGE_SANITIZE AND EDGE_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
include(cmake/edge-options.cmake)
find_package(Threads REQUIRED)
file(GLOB PROJECT_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(telemetry_batch ${PROJECT_SOURCES})
target_include_directories(telemetry_batch PRIVATE include)
target_link_libraries(telemetry_batch PRIVATE Threads::Threads)
edge_configure_target(telemetry_batch)
target_compile_options(telemetry_batch PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-omit-frame-pointer)
if(EDGE_SANITIZE)
  target_compile_options(telemetry_batch PRIVATE -O1 -g -fsanitize=address,undefined)
  target_link_options(telemetry_batch PRIVATE -fsanitize=address,undefined)
endif()
if(EDGE_TSAN)
  target_compile_options(telemetry_batch PRIVATE -O1 -g -fsanitize=thread)
  target_link_options(telemetry_batch PRIVATE -fsanitize=thread)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME correctness COMMAND telemetry_batch --test)
  add_test(NAME benchmark_smoke COMMAND telemetry_batch --mode all --samples 2 --warmup 1)
  add_test(NAME reject_unknown_option COMMAND telemetry_batch --invalid)
  add_test(NAME reject_zero_samples COMMAND telemetry_batch --samples 0)
  set_tests_properties(reject_unknown_option reject_zero_samples PROPERTIES WILL_FAIL TRUE)
  set_tests_properties(correctness benchmark_smoke PROPERTIES TIMEOUT 90)
endif()
