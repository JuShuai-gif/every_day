cmake_minimum_required(VERSION 3.18)
project(cache_counters LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SANITIZER "" CACHE STRING "address or thread")
find_package(Threads REQUIRED)
add_executable(cache_counters src/main.cpp src/hot_loop.cpp)
target_link_libraries(cache_counters PRIVATE Threads::Threads)
target_compile_options(cache_counters PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(SANITIZER STREQUAL "address")
  target_compile_options(cache_counters PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cache_counters PRIVATE -fsanitize=address,undefined)
elseif(SANITIZER STREQUAL "thread")
  target_compile_options(cache_counters PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(cache_counters PRIVATE -fsanitize=thread)
elseif(NOT SANITIZER STREQUAL "")
  message(FATAL_ERROR "Unknown SANITIZER")
endif()
