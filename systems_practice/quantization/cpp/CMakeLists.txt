cmake_minimum_required(VERSION 3.16)
project(quant_cpu LANGUAGES CXX)
add_executable(quant_cpu src/quant_cpu.cpp)
target_compile_features(quant_cpu PRIVATE cxx_std_17)
target_include_directories(quant_cpu PRIVATE ../../common/cpp)
target_compile_options(quant_cpu PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
option(SANITIZE "Enable ASan and UBSan" OFF)
if(SANITIZE)
  target_compile_options(quant_cpu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(quant_cpu PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME storage_boundaries COMMAND quant_cpu --self-test)
