cmake_minimum_required(VERSION 3.18)
project(ptx_token_pool LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_CUDA "Build Jetson Thor SM110 pooling variants" OFF)
option(ENABLE_SANITIZERS "Check CPU reference with ASan/UBSan" OFF)
add_executable(cpu_check src/cpu_check.cpp)
target_compile_options(cpu_check PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
  target_compile_options(cpu_check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cpu_check PRIVATE -fsanitize=address,undefined)
endif()
if(ENABLE_CUDA)
  # 所有 CUDA/PTX/SASS 固定针对用户指定的 Thor SM110，不静默回退或切换后缀。
  if(DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "110")
    message(FATAL_ERROR "This practice is fixed to Jetson Thor SM110; use CMAKE_CUDA_ARCHITECTURES=110")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES 110)
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
    message(FATAL_ERROR "This SM110 project selects CUDA Toolkit 13.0 or newer")
  endif()
  add_executable(ptx_pool src/ptx_pool.cu)
  set_target_properties(ptx_pool PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(ptx_pool PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
enable_testing()
add_test(NAME cpu_contract COMMAND cpu_check)
