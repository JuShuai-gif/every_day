cmake_minimum_required(VERSION 3.20)
project(thor_reduce LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_CUDA "Build Thor SM110 target" OFF)
option(SANITIZE "ASan/UBSan for CPU contracts" OFF)
add_executable(cpu_check src/cpu_check.cpp)
target_compile_options(cpu_check PRIVATE -Wall -Wextra -Wpedantic)
if(SANITIZE)
  target_compile_options(cpu_check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cpu_check PRIVATE -fsanitize=address,undefined)
endif()
if(ENABLE_CUDA)
  if(DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "110")
    message(FATAL_ERROR "Only Thor SM110 is permitted")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES 110 CACHE STRING "Fixed Thor target" FORCE)
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
    message(FATAL_ERROR "Thor SM110 requires CUDA Toolkit 13.0+")
  endif()
  add_executable(row_reduce src/row_reduce.cu)
  set_target_properties(row_reduce PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(row_reduce PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
