cmake_minimum_required(VERSION 3.16)
project(paper_example LANGUAGES CXX)
add_executable(paper_example src/example.cpp)
target_compile_features(paper_example PRIVATE cxx_std_17)
target_include_directories(paper_example PRIVATE ../../../common/cpp)
target_compile_options(paper_example PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
option(SANITIZE "Enable ASan and UBSan" OFF)
if(SANITIZE)
  target_compile_options(paper_example PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(paper_example PRIVATE -fsanitize=address,undefined)
endif()
