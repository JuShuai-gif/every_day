cmake_minimum_required(VERSION 3.20)
project(q8_block_dot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SANITIZE "Enable ASan and UBSan" OFF)
option(BASE_ARM "Compile base Armv8-A without optional dot product" OFF)
add_executable(q8_dot src/main.cpp src/scalar.cpp src/neon.cpp)
target_compile_options(q8_dot PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  set_source_files_properties(src/scalar.cpp PROPERTIES COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  set_source_files_properties(src/scalar.cpp PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize")
endif()
if(BASE_ARM)
  target_compile_options(q8_dot PRIVATE -march=armv8-a)
endif()
if(SANITIZE)
  target_compile_options(q8_dot PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(q8_dot PRIVATE -fsanitize=address,undefined)
endif()
