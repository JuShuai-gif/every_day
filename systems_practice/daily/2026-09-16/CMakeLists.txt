cmake_minimum_required(VERSION 3.20)
project(tensorrt_dynamic_practice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_TENSORRT "Build the actual NVIDIA GPU implementation" OFF)
option(ENABLE_SANITIZERS "CPU AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(reference STATIC src/common.cpp)
target_include_directories(reference PUBLIC src)
target_compile_options(reference PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cpu_check src/cpu_check.cpp)
target_link_libraries(cpu_check PRIVATE reference)
target_compile_options(cpu_check PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
  target_compile_options(reference PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(reference PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME cpu_reference COMMAND cpu_check)
if(ENABLE_TENSORRT)
  find_package(CUDAToolkit REQUIRED)
  set(TENSORRT_ROOT "" CACHE PATH "Existing TensorRT installation root")
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h
    HINTS "${TENSORRT_ROOT}" PATH_SUFFIXES include include/x86_64-linux-gnu include/aarch64-linux-gnu)
  find_library(TENSORRT_LIBRARY nvinfer
    HINTS "${TENSORRT_ROOT}" PATH_SUFFIXES lib lib64 lib/x86_64-linux-gnu lib/aarch64-linux-gnu)
  if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY)
    message(FATAL_ERROR "TensorRT headers/libnvinfer missing: set TENSORRT_ROOT or explicit TENSORRT_INCLUDE_DIR/TENSORRT_LIBRARY")
  endif()
  add_executable(trt_dynamic src/trt_dynamic.cpp)
  target_include_directories(trt_dynamic PRIVATE "${TENSORRT_INCLUDE_DIR}")
  target_link_libraries(trt_dynamic PRIVATE reference "${TENSORRT_LIBRARY}" CUDA::cudart)
  target_compile_options(trt_dynamic PRIVATE -Wall -Wextra -Wpedantic)
endif()
