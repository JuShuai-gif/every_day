cmake_minimum_required(VERSION 3.24)
project(thor_energy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cpu src/cpu.cpp)
option(ENABLE_CUDA "Build Thor" OFF)
if(ENABLE_CUDA)
 set(CMAKE_CUDA_ARCHITECTURES 110 CACHE STRING "Thor only" FORCE)
 enable_language(CUDA)
 if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
  message(FATAL_ERROR "Use Thor-compatible CUDA 13+ toolkit")
 endif()
 add_executable(energy src/energy.cu)
 set_target_properties(energy PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
 target_compile_options(energy PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
