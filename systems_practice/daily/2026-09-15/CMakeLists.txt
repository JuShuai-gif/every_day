cmake_minimum_required(VERSION 3.18)
project(dual_camera_preprocess LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cpu_check src/cpu_check.cpp)
target_compile_options(cpu_check PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME cpu_correctness COMMAND cpu_check)
option(PRACTICE_ENABLE_CUDA "Build CUDA example if the compiler exists" ON)
if(PRACTICE_ENABLE_CUDA)
    include(CheckLanguage)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        enable_language(CUDA)
        add_executable(cuda_preprocess src/preprocess.cu)
        set_target_properties(cuda_preprocess PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
        target_compile_options(cuda_preprocess PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
    else()
        message(STATUS "CUDA compiler missing: only CPU reference can be verified")
    endif()
endif()
