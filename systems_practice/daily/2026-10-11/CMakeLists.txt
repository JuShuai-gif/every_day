cmake_minimum_required(VERSION 3.20)
project(dynamic_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(cpu_check src/cpu_check.cpp)
option(THOR "Build only for Thor SM110" OFF)
if(THOR)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_ARCHITECTURES 110)
  add_executable(kernels src/kernels.cu)
  target_compile_options(kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  find_path(TRT_INCLUDE NvInfer.h HINTS ENV TRT_ROOT PATH_SUFFIXES include include/aarch64-linux-gnu)
  find_library(TRT_LIBRARY nvinfer HINTS ENV TRT_ROOT PATH_SUFFIXES lib lib64 lib/aarch64-linux-gnu)
  if(NOT TRT_INCLUDE OR NOT TRT_LIBRARY)
    message(FATAL_ERROR "TensorRT headers/shared library missing")
  endif()
  add_executable(trt src/trt.cpp)
  target_include_directories(trt PRIVATE ${TRT_INCLUDE})
  target_link_libraries(trt PRIVATE ${TRT_LIBRARY} CUDA::cudart)
endif()
