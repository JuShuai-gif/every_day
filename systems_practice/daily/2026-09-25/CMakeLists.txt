cmake_minimum_required(VERSION 3.18)
project(rollback LANGUAGES CXX)
add_executable(rollback src/main.cpp)
target_compile_features(rollback PRIVATE cxx_std_17)
target_compile_options(rollback PRIVATE -Wall -Wextra -Werror)
if(SANITIZER)
 target_compile_options(rollback PRIVATE -fsanitize=${SANITIZER} -fno-omit-frame-pointer)
 target_link_options(rollback PRIVATE -fsanitize=${SANITIZER})
endif()
