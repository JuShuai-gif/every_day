cmake_minimum_required(VERSION 3.16)
project(edge_deployment LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(edge_deployment src/main.cpp)
target_compile_features(edge_deployment PRIVATE cxx_std_17)
target_compile_options(edge_deployment PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(edge_deployment PRIVATE Threads::Threads)
set(SANITIZER "" CACHE STRING "address,undefined or thread")
if(SANITIZER)
  target_compile_options(edge_deployment PRIVATE -fsanitize=${SANITIZER} -fno-omit-frame-pointer)
  target_link_options(edge_deployment PRIVATE -fsanitize=${SANITIZER})
endif()
