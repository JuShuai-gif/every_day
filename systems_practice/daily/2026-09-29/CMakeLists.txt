cmake_minimum_required(VERSION 3.18)
project(rknn_output_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(cpu_check src/cpu_check.cpp)
target_compile_options(cpu_check PRIVATE -Wall -Wextra -Wpedantic)
option(SANITIZE "CPU ASan/UBSan" OFF)
if(SANITIZE)
  target_compile_options(cpu_check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cpu_check PRIVATE -fsanitize=address,undefined)
endif()
option(BUILD_RKNN "Build on RK3588 Linux with installed Runtime" OFF)
if(BUILD_RKNN)
  find_path(RKNN_INCLUDE rknn_api.h HINTS "$ENV{RKNN_ROOT}/include" REQUIRED)
  find_library(RKNN_LIBRARY rknnrt HINTS "$ENV{RKNN_ROOT}/lib" "$ENV{RKNN_ROOT}/lib64" REQUIRED)
  add_executable(rknn_output src/rknn_output.cpp)
  target_include_directories(rknn_output PRIVATE "${RKNN_INCLUDE}")
  target_link_libraries(rknn_output PRIVATE "${RKNN_LIBRARY}")
endif()
