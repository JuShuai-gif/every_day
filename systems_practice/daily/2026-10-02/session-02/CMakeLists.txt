cmake_minimum_required(VERSION 3.24)
project(thor_transpose LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_CUDA "Build Thor target" OFF)
add_executable(cpu src/cpu.cpp)
if(ENABLE_CUDA)
  set(CMAKE_CUDA_ARCHITECTURES 110 CACHE STRING "Thor only" FORCE)
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
    message(FATAL_ERROR "Thor requires compatible CUDA 13+ / JetPack 7 environment")
  endif()
  add_executable(transpose src/transpose.cu)
  set_target_properties(transpose PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(transpose PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
