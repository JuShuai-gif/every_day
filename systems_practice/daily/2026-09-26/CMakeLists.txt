cmake_minimum_required(VERSION 3.24)
project(thor_softmax LANGUAGES CXX)
add_executable(cpu src/cpu.cpp)
target_compile_features(cpu PRIVATE cxx_std_17)
if(SANITIZER)
 target_compile_options(cpu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cpu PRIVATE -fsanitize=address,undefined)
endif()
option(ENABLE_CUDA "Build Thor SM110 candidate" OFF)
if(ENABLE_CUDA)
 enable_language(CUDA)
 add_executable(softmax src/softmax.cu)
 set_target_properties(softmax PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES CUDA_ARCHITECTURES 110)
 target_compile_options(softmax PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
