cmake_minimum_required(VERSION 3.18)
project(thor_quant_gemm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_CUDA "Build Thor SM110 quantized GEMM" OFF)
option(ENABLE_SANITIZERS "CPU ASan/UBSan checks" OFF)
add_executable(quant_cpu src/main.cpp)
target_compile_options(quant_cpu PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
  target_compile_options(quant_cpu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(quant_cpu PRIVATE -fsanitize=address,undefined)
endif()
if(ENABLE_CUDA)
  if(DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "110")
    message(FATAL_ERROR "This project targets Thor SM110 only")
  endif()
  set(CMAKE_CUDA_ARCHITECTURES 110)
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
    message(FATAL_ERROR "This project selects CUDA Toolkit 13.0+")
  endif()
  add_executable(quant_gpu src/quant_gemm.cu)
  set_target_properties(quant_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(quant_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
