cmake_minimum_required(VERSION 3.16)
project(bounded_inference_pool LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
set(SANITIZER "none" CACHE STRING "none, address, thread")
add_executable(pool_check src/main.cpp)
target_link_libraries(pool_check PRIVATE Threads::Threads)
target_compile_options(pool_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(SANITIZER STREQUAL "address")
  target_compile_options(pool_check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pool_check PRIVATE -fsanitize=address,undefined)
elseif(SANITIZER STREQUAL "thread")
  target_compile_options(pool_check PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(pool_check PRIVATE -fsanitize=thread)
elseif(NOT SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown sanitizer")
endif()
enable_testing()
add_test(NAME concurrency_contract COMMAND pool_check check)
set_tests_properties(concurrency_contract PROPERTIES TIMEOUT 45)
